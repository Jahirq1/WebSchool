===============Lajmet.php=================tabel per shtimin e lajmev ne sliderin e page te lajmeve======
CREATE TABLE post (
    id INT AUTO_INCREMENT PRIMARY KEY,     -- Identifikues unik
    image_url VARCHAR(255) NOT NULL,       -- URL-ja e imazhit
    title VARCHAR(255) NOT NULL,           -- Titulli i lajmit
    content TEXT NOT NULL,                 -- Përmbajtja e lajmit
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP -- Koha kur është krijuar postimi
);
===============Eventet.php=================
CREATE TABLE `eventet` (
  `id` INT(11) AUTO_INCREMENT PRIMARY KEY,
  `title` VARCHAR(255) NOT NULL,
  `content` TEXT NOT NULL,
  `image` VARCHAR(255) NOT NULL,
  `date` DATE NOT NULL,
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
